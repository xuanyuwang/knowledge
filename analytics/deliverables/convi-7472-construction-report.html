What the change must do

Trigger and input

A user opens the QM Appeals report or changes its filters. The chart receives appeal statistics grouped by time range, including separate appealed, resolved and adjusted conversation counts.

Expected result

Each Adjusted bar uses that bucket’s adjustedConversationCount. A zero or omitted count keeps the existing zero fallback. Appealed and Resolved keep their own counts.

Here “adjusted” is the API’s conversation-level count for resolved appeals with an adjustment. The By Criteria overview measures criteria instead: its percentage uses adjusted criteria ÷ resolved criteria. These are different units.

One production line changes. API requests, date filtering, backend aggregation, table rows and the criteria overview retain their existing behavior. The separate requested adjusted-appeals ÷ all-evaluations ratio is outside this fix.

The data path

1 · Report filtersDate range, frequency, scorecard, reviewer and user/team/group selections become the request.
2 · Appeal statisticsThe real query hook adds use-case context and retrieves time buckets from the API.
3 · Series mappingConvert bucket timestamps, keep visible dates and map each conversation count. Only the Adjusted array reference changes.
4 · Chart optionsHighchartsWithSpinner receives the three series. Tests observe these final options.

This display path reads data. It does not write customer records. Server authorization is upstream and is not re-evaluated by this one-line frontend change.

import type { ReactElement } from 'react';
import { memo, useMemo } from 'react';
import { Paper } from '@mantine/core';
import dayjs from 'dayjs';
import type { SeriesColumnOptions } from 'highcharts';
import { merge } from 'lodash-es';

import type { DateRange } from '@cresta/director-api';
import { createDatesInRangeForFrequency } from '@cresta/director-utils';
import { QAAttributeType } from '@cresta/web-client';

import { HighchartsWithSpinner } from '@/components/spinners';
import { useRetrieveAppealStats } from '@/hooks/qa';
import { useTranslation } from '@/hooks/useTranslation';
import {
  getDefaultDateTimeXAxisOptions,
  getTickIntervalForMaxValue,
  getTickIntervalForTimeRange,
  mapDateToHighcharts,
  notUndefined,
} from '@/utils';

import { useDefaultQAChartOptions } from '../../../highchartsUtils';
import type { QAReportFiltersState } from '../../../hooks';
import { transformFilterStateToRetrieveAppealStatsRequest } from '../../../shared';

import styles from './QAReportAppealTrendsCard.module.css';

const MAX_DATE_RANGES = 8;

export const QAReportAppealTrendsCard = memo(function QAReportAppealTrendsCard(props: {
  filterState: QAReportFiltersState;
}): ReactElement {
  const { t } = useTranslation('director-app-qm', { keyPrefix: 'report.appeals.trend-chart' });
  const appealStats = useRetrieveAppealStats({
    ...transformFilterStateToRetrieveAppealStatsRequest(props.filterState),
    groupByAttributes: [QAAttributeType.QA_ATTRIBUTE_TYPE_TIME_RANGE],
  });

  const defaultQAChartOptions = useDefaultQAChartOptions();

  const chartOptions = useMemo((): Highcharts.Options => {
    const dateRanges = createDatesInRangeForFrequency(
      props.filterState.selectedDateRange.startDate,
      props.filterState.selectedDateRange.endDate,
      props.filterState.frequency
    );
    const startDate = dateRanges.slice(-MAX_DATE_RANGES)[0];
    const endDate = dateRanges[dateRanges.length - 1];
    if (!startDate || !endDate) {
      return defaultQAChartOptions;
    }
    const dateRange = [dateRanges.slice(-MAX_DATE_RANGES)[0], dateRanges[dateRanges.length - 1]] as DateRange;
    const lastNDates = new Set(dateRanges.slice(-MAX_DATE_RANGES).map(mapDateToHighcharts));

    // Fill in the actual data
    const appealedData = appealStats.data?.appealStats
      ?.map((stat) => {
        const highchartsDate = mapDateToHighcharts(dayjs(stat.attribute?.timeRange?.startTimestamp).tz());
        if (!stat.attribute?.timeRange?.startTimestamp || !lastNDates.has(highchartsDate)) {
          return;
        }
        return [highchartsDate, stat.appealedConversationCount || 0] as const;
      })
      .filter(notUndefined);

    const resolvedData = appealStats.data?.appealStats
      ?.map((stat) => {
        const highchartsDate = mapDateToHighcharts(dayjs(stat.attribute?.timeRange?.startTimestamp).tz());
        if (!stat.attribute?.timeRange?.startTimestamp || !lastNDates.has(highchartsDate)) {
          return;
        }
        return [highchartsDate, stat.resolvedConversationCount || 0] as const;
      })
      .filter(notUndefined);

    const adjustedData = appealStats.data?.appealStats
      ?.map((stat) => {
        const highchartsDate = mapDateToHighcharts(dayjs(stat.attribute?.timeRange?.startTimestamp).tz());
        if (!stat.attribute?.timeRange?.startTimestamp || !lastNDates.has(highchartsDate)) {
          return;
        }
        return [highchartsDate, stat.adjustedConversationCount || 0] as const;
      })
      .filter(notUndefined);

    // Calculate max value for ticks
    let maxYValue = 0;
    appealedData?.forEach(([_, val]) => (maxYValue = Math.max(maxYValue, val)));
    resolvedData?.forEach(([_, val]) => (maxYValue = Math.max(maxYValue, val)));
    adjustedData?.forEach(([_, val]) => (maxYValue = Math.max(maxYValue, val)));

    const tickInterval = getTickIntervalForTimeRange(dateRange, props.filterState.frequency);

    const appealedSeries: SeriesColumnOptions = {
      type: 'column',
      data: appealedData,
      color: 'var(--border-default)',
      name: t('series.appealed', 'Appealed'),
      zoneAxis: 'x',
      pointRange: tickInterval,
    };

    const resolvedSeries: SeriesColumnOptions = {
      type: 'column',
      data: resolvedData,
      color: 'var(--content-action)',
      name: t('series.resolved', 'Resolved'),
      zoneAxis: 'x',
      pointRange: tickInterval,
    };

    const adjustedSeries: SeriesColumnOptions = {
      type: 'column',
      data: adjustedData,
      color: 'var(--extended-content-orange)',
      name: t('series.adjusted', 'Adjusted'),
      zoneAxis: 'x',
      pointRange: tickInterval,
    };

    return merge({}, defaultQAChartOptions, {
      title: {
        text: `<span style="font-size: 18px; font-weight: 500; font-family: 'Inter', sans-serif;">${t(
          'title',
          'Appeals Over Time'
        )}</span>`,
      },
      series: [appealedSeries, resolvedSeries, adjustedSeries],
      xAxis: {
        ...getDefaultDateTimeXAxisOptions(),
        tickInterval: getTickIntervalForTimeRange(dateRange, props.filterState.frequency),
      },
      yAxis: {
        max: maxYValue + 1,
        tickInterval: getTickIntervalForMaxValue(maxYValue),
        title: {
          text: t('axis.appeals', '# of appeals'),
        },
      },
      lang: {
        noData: t('no-data', 'No data to display'),
      },
    });
  }, [
    appealStats.data?.appealStats,
    props.filterState.frequency,
    props.filterState.selectedDateRange.endDate,
    props.filterState.selectedDateRange.startDate,
    defaultQAChartOptions,
    t,
  ]);

  return (
    <Paper
      data-testid="appeals-trends-chart"
      h={200}
      radius="lg"
      shadow="subtle"
      p="lg"
      flex={1}
      className={styles.cardShell}
    >
      <HighchartsWithSpinner
        loading={appealStats.isLoading}
        options={chartOptions}
        chartDescription={t(
          'description',
          'Interactive chart showing the number of appeals and evaluated conversations over time'
        )}
      />
    </Paper>
  );
});

import { waitFor } from '@testing-library/react';
import dayjs from 'dayjs';
import { afterEach, describe, expect, it, vi } from 'vitest';

import { CrestaAPI } from '@cresta/director-api';
import type { RetrieveAppealStatsResponse } from '@cresta/web-client';

import { HighchartsWithSpinner } from '@/components/spinners';
import { render } from '@/testing';

import type { QAReportFiltersState } from '../../../hooks';

import { QAReportAppealTrendsCard } from './QAReportAppealTrendsCard';

vi.mock('@/components/spinners', () => ({ HighchartsWithSpinner: vi.fn(() => null) }));

function setup(adjustedConversationCount: number | undefined): void {
  const response: RetrieveAppealStatsResponse = {
    appealStats: [
      {
        attribute: { timeRange: { startTimestamp: '2026-07-01T00:00:00Z' } },
        appealedConversationCount: 10,
        resolvedConversationCount: 10,
        adjustedConversationCount,
        appealedCriteriaCount: 12,
        resolvedCriteriaCount: 12,
        adjustedCriteriaCount: 7,
        appealedConversations: [],
      },
    ],
  };
  vi.spyOn(CrestaAPI.insights, 'retrieveAppealStats').mockReturnValue({
    promise: Promise.resolve(response),
    cancel: vi.fn(),
  });

  const selectedDateRange = {
    startDate: dayjs('2026-07-01T00:00:00Z').tz(),
    endDate: dayjs('2026-07-31T00:00:00Z').tz(),
  };
  const filterState: QAReportFiltersState = {
    scorecardTemplateResourceName: undefined,
    selectedDateRange,
    selectedDateRangeInternal: selectedDateRange,
    frequency: 'MONTHLY',
    evaluationPeriods: [],
    useCustomDateRange: true,
    useFlexibleDateRange: false,
    usersTeamsGroups: { userNames: [], teamNames: [], groupNames: [] },
  };

  render(<QAReportAppealTrendsCard filterState={filterState} />);
}

describe('QAReportAppealTrendsCard', () => {
  afterEach(() => {
    vi.restoreAllMocks();
    vi.clearAllMocks();
  });

  it.each([
    { adjustedConversationCount: 3, expectedAdjustedCount: 3 },
    { adjustedConversationCount: 0, expectedAdjustedCount: 0 },
    { adjustedConversationCount: undefined, expectedAdjustedCount: 0 },
  ])(
    'plots $expectedAdjustedCount adjusted appeals when the API returns $adjustedConversationCount',
    async ({ adjustedConversationCount, expectedAdjustedCount }) => {
      setup(adjustedConversationCount);

      await waitFor(() => {
        expect(vi.mocked(HighchartsWithSpinner).mock.lastCall?.[0].options?.series).toEqual([
          expect.objectContaining({ name: 'Appealed', data: [[expect.any(Number), 10]] }),
          expect.objectContaining({ name: 'Resolved', data: [[expect.any(Number), 10]] }),
          expect.objectContaining({ name: 'Adjusted', data: [[expect.any(Number), expectedAdjustedCount]] }),
        ]);
      });
    }
  );
});
